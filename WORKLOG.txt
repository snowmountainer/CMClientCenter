<#
    Test-CCMProgram.ps1   (v2 — ersetzt Test-CCMTaskSequence.ps1)

    HINTERGRUND
    -----------
    Der erste Test hat gezeigt: CCM_TaskSequence existiert auf diesem
    Client NICHT, obwohl das native Software Center die OSD-TS
    "OSD Windows 11 Enterprise 25H2" unter "Operating Systems" anzeigt.

    Recherche-Ergebnis: Task Sequences (inkl. OSD) werden im Client SDK
    nicht ueber CCM_TaskSequence, sondern ueber CCM_Program dargestellt
    (ROOT\ccm\clientSDK). CCM_Program hat ein Bool-Property "TaskSequence",
    das bei TS-Items auf True steht. CCM_Program deckt also sowohl normale
    Programs als auch Task-Sequence-Items ab.

    ZWECK DIESES SCRIPTS
    ---------------------
    Reine Diagnose, wie zuvor. Es fuehrt NICHTS aus (kein Invoke-CimMethod).
    Es prueft:
      1. Existiert CCM_Program?
      2. Welche Properties hat es (Schema)?
      3. Welche Methoden bietet die Klasse (Schema) - fuer den spaeteren
         Invoke-Schritt, damit wir die Methode nicht raten muessen.
      4. Alle aktuellen Instanzen, ungefiltert, inkl. TaskSequence-Flag.

    AUSFUEHRUNG
    -----------
    Direkt auf dem Testclient als Administrator, PowerShell 5.1:

        .\Test-CCMProgram.ps1
        .\Test-CCMProgram.ps1 -ExportPath C:\Temp\ccmprogram_dump.json

    Bitte komplette Ausgabe zurueck in den Chat geben.
#>

[CmdletBinding()]
param(
    [string]$ExportPath
)

$ErrorActionPreference = 'Stop'

Write-Host "=== CCM_Program Diagnose (v2) ===" -ForegroundColor Cyan
Write-Host "Host: $env:COMPUTERNAME"
Write-Host "Datum: $(Get-Date -Format 'yyyy-MM-dd HH:mm:ss')"
Write-Host ""

# --- 1. Existiert die Klasse? ---------------------------------------------
try {
    $class = Get-CimClass -Namespace 'ROOT\ccm\clientsdk' -ClassName 'CCM_Program' -ErrorAction Stop
    Write-Host "[OK] Klasse CCM_Program in ROOT\ccm\clientsdk gefunden." -ForegroundColor Green
} catch {
    Write-Host "[FEHLER] Klasse CCM_Program nicht gefunden oder Namespace nicht erreichbar." -ForegroundColor Red
    Write-Host "Meldung: $($_.Exception.Message)" -ForegroundColor Red
    return
}

# --- 2. Properties-Schema --------------------------------------------------
Write-Host ""
Write-Host "--- Schema: Properties von CCM_Program ---" -ForegroundColor Cyan
$class.CimClassProperties.Name | Sort-Object | ForEach-Object { Write-Host "  - $_" }

# --- 3. Methoden-Schema (wichtig fuer spaeteren Invoke-Schritt) ------------
Write-Host ""
Write-Host "--- Schema: Methoden von CCM_Program ---" -ForegroundColor Cyan
foreach ($m in $class.CimClassMethods) {
    $paramList = ($m.Parameters | ForEach-Object { "$($_.CimType) $($_.Name)" }) -join ", "
    Write-Host "  - $($m.Name)($paramList)"
}

# --- 4. Instanzen: alle aktuell verfuegbaren Programs/Task Sequences ------
Write-Host ""
Write-Host "--- Instanzen: aktuell verfuegbare Programs (inkl. Task Sequences) ---" -ForegroundColor Cyan

$instances = @(Get-CimInstance -Namespace 'ROOT\ccm\clientsdk' -ClassName 'CCM_Program' -ErrorAction SilentlyContinue)

if ($instances.Count -eq 0) {
    Write-Host "[INFO] Keine Programs/Task Sequences fuer diesen Client verfuegbar." -ForegroundColor Yellow
    Write-Host "Das widerspricht dem, was im Software Center sichtbar war - bitte pruefen," -ForegroundColor Yellow
    Write-Host "ob dieses Script als Administrator/SYSTEM-Kontext laeuft (siehe Hinweis zu" -ForegroundColor Yellow
    Write-Host "CCM_Program-Zugriffsproblemen fuer Standardbenutzer)." -ForegroundColor Yellow
} else {
    Write-Host "[OK] $($instances.Count) Eintrag/Eintraege gefunden." -ForegroundColor Green
}

$dump = @()

foreach ($prg in $instances) {
    Write-Host ""
    Write-Host ("-" * 70)
    $props = [ordered]@{}
    foreach ($p in $prg.CimInstanceProperties) {
        $name = $p.Name
        $value = $p.Value
        $display = if ($value -is [System.Array]) { ($value -join '; ') } else { $value }
        $props[$name] = $display
        Write-Host ("{0,-30}: {1}" -f $name, $display)
    }
    $dump += [PSCustomObject]$props
}

# --- 5. Direkter Filter auf TaskSequence = True, zur Bestaetigung ----------
Write-Host ""
Write-Host "--- Filter: nur Eintraege mit TaskSequence = True ---" -ForegroundColor Cyan
$tsOnly = @($instances | Where-Object { $_.TaskSequence -eq $true })
if ($tsOnly.Count -eq 0) {
    Write-Host "[INFO] Kein Eintrag mit TaskSequence=True gefunden." -ForegroundColor Yellow
    Write-Host "Falls oben trotzdem 'OSD Windows 11...' in der Liste aller Instanzen steht," -ForegroundColor Yellow
    Write-Host "pruefen wir anhand der rohen Properties, welches Feld stattdessen TS markiert." -ForegroundColor Yellow
} else {
    Write-Host "[OK] $($tsOnly.Count) Task-Sequence-Eintrag/Eintraege erkannt:" -ForegroundColor Green
    foreach ($t in $tsOnly) { Write-Host "  - $($t.Name)" }
}

# --- 6. Export --------------------------------------------------------------
if ($ExportPath) {
    try {
        $dump | ConvertTo-Json -Depth 5 | Out-File -FilePath $ExportPath -Encoding UTF8
        Write-Host ""
        Write-Host "[OK] Export gespeichert: $ExportPath" -ForegroundColor Green
    } catch {
        Write-Host ""
        Write-Host "[FEHLER] Export fehlgeschlagen: $($_.Exception.Message)" -ForegroundColor Red
    }
}

Write-Host ""
Write-Host "=== Ende Diagnose ===" -ForegroundColor Cyan
Write-Host "Bitte komplette Ausgabe zurueck in den Chat geben."
